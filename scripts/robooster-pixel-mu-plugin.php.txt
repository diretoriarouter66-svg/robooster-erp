add_action( 'wp_footer', function () {
  if ( is_admin() || is_user_logged_in() ) return;
  $meta_view_content = '';
  if ( function_exists( 'is_product' ) && is_product() ) {
    $meta_view_content = "fbq('track','ViewContent',{content_name:'" . esc_js( get_the_title() ) . "',content_type:'product'});";
  }
  ?>
  <script>
  // ===== Pixel da Meta (conjunto de dados Robooster Site) =====
  (function(){
    try {
      if (localStorage.getItem('rb_optout') === '1') return;
      !function(f,b,e,v,n,t,s){if(f.fbq)return;n=f.fbq=function(){n.callMethod?
      n.callMethod.apply(n,arguments):n.queue.push(arguments)};if(!f._fbq)f._fbq=n;
      n.push=n;n.loaded=!0;n.version='2.0';n.queue=[];t=b.createElement(e);t.async=!0;
      t.src=v;s=b.getElementsByTagName(e)[0];s.parentNode.insertBefore(t,s)}(window,
      document,'script','https://connect.facebook.net/en_US/fbevents.js');
      fbq('init', '1547298694107494');
      fbq('track', 'PageView');
      <?php echo $meta_view_content; ?>
      document.addEventListener('click', function(e){
        var a = e.target.closest && e.target.closest('a[href*="wa.me"], a[href*="api.whatsapp.com"]');
        if (a && window.fbq) fbq('track', 'Contact');
      }, true);
    } catch(err) {}
  })();
  </script>
  <script>
  (function(){
    try {
      if (localStorage.getItem('rb_optout') === '1') return;
      var m = document.cookie.match(/(?:^|; )rb_vid=([^;]+)/);
      var vid = m ? m[1] : (Date.now().toString(36) + Math.random().toString(36).slice(2, 10));
      if (!m) document.cookie = 'rb_vid=' + vid + '; max-age=34186698; path=/; SameSite=Lax';
      var qs = new URLSearchParams(location.search);
      var utm = {};
      ['utm_source','utm_medium','utm_campaign','utm_term','utm_content'].forEach(function(k){ if (qs.get(k)) utm[k] = qs.get(k); });
      var send = function(evento){
        var payload = JSON.stringify({
          visitor_id: vid, evento: evento, url: location.pathname,
          titulo: document.title.slice(0,180), referrer: document.referrer.slice(0,300),
          utm: Object.keys(utm).length ? utm : null, user_agent: navigator.userAgent.slice(0,200)
        });
        navigator.sendBeacon('https://supabase.robooster.com.br/rest/v1/site_eventos?apikey=eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.ewogICJyb2xlIjogImFub24iLAogICJpc3MiOiAic3VwYWJhc2UiLAogICJpYXQiOiAxNzE1MDUwODAwLAogICJleHAiOiAxODcyODE3MjAwCn0.1ipRkm26SsMllqrEG_142hoCRXeQ1vfJd-oyAx1p9b8',
          new Blob([payload], {type: 'application/json'}));
      };
      send('pageview');
      document.addEventListener('click', function(e){
        var a = e.target.closest && e.target.closest('a[href*="wa.me"], a[href*="api.whatsapp.com"]');
        if (!a) return;
        send('whatsapp_click');
        try {
          // Assinatura do visitante na mensagem: liga a navegação ao WhatsApp dele
          var u = new URL(a.href);
          var ref = ' (ref ' + vid.slice(0, 8) + ')';
          var t = u.searchParams.get('text') || 'Ol\u00e1! Vim pelo site da Robooster.';
          if (t.indexOf('(ref ') === -1) { u.searchParams.set('text', t + ref); a.href = u.toString(); }
        } catch(err) {}
      }, true);
    } catch(e) {}
  })();
  </script>
  <?php
} );
